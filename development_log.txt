Actual Model: Used is_archived field instead

Impact: /api/v1/documents/ endpoint returning 500 errors

Root Cause Analysis
1. Model Definition (employees/models/document.py)
class Document(models.Model):
    # ... other fields ...
    is_archived = models.BooleanField(default=False)
    # NO verified_by or is_verified fields existed
2. Serializer Mismatch (employees/serializers.py)
class DocumentSerializer(serializers.ModelSerializer):
    class Meta:
        fields = [
            # ... other fields ...
            'verified_by',      # ❌ Field doesn't exist
            'is_verified',      # ❌ Field doesn't exist
            # 'is_archived' was missing ✅
        ]
3. ViewSet Logic Issue (employees/views.py)
class DocumentViewSet(viewsets.ModelViewSet):
    def perform_create(self, serializer):
        serializer.save(
            verified_by=self.request.user  # ❌ Field doesn't exist
        )

Solution Implemented
Step 1: Updated DocumentSerializer
# File: hr_core/employees/serializers.py
class DocumentSerializer(serializers.ModelSerializer):
    employee_name = serializers.CharField(source='employee.get_full_name', read_only=True)
    
    class Meta:
        model = Document
        fields = [
            'id', 'employee', 'employee_name', 'title', 
            'document_type', 'file_path', 'file_size', 
            'mime_type', 'uploaded_at', 'uploaded_by',
            'is_archived'  # ✅ Added correct field
        ]
        read_only_fields = ['id', 'uploaded_at', 'file_size', 'mime_type']
Step 2: Cleaned Up DocumentViewSet
# File: hr_core/employees/views.py
class DocumentViewSet(viewsets.ModelViewSet):
    queryset = Document.objects.select_related('employee').all()
    serializer_class = DocumentSerializer
    
    def get_queryset(self):
        queryset = super().get_queryset()
        employee_id = self.request.query_params.get('employee')
        document_type = self.request.query_params.get('document_type')
        is_archived = self.request.query_params.get('is_archived')  # ✅ Updated filter
        
        if employee_id:
            queryset = queryset.filter(employee_id=employee_id)
        if document_type:
            queryset = queryset.filter(document_type=document_type)
        if is_archived is not None:  # ✅ Updated filter logic
            queryset = queryset.filter(is_archived=is_archived.lower() == 'true')
            
        return queryset
    
    def perform_create(self, serializer):
        serializer.save(uploaded_by=self.request.user if self.request.user.is_authenticated else None)
        # ✅ Removed verified_by assignment
